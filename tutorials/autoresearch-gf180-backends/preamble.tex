% Right-flushed tool banner. Borrowed from rtl2gds-gf180-docker.
% Usage: \driving{LibreLane}  (place right after \begin{frame}{...}).
\newcommand{\driving}[1]{%
  \begin{flushright}%
    \vspace{-0.9em}%
    \colorbox{accent!10}{%
      \footnotesize\color{accent}%
      Driving: \textbf{#1}%
    }%
  \end{flushright}%
  \vspace{-0.4em}%
}
