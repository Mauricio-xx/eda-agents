\section{The idea\_to\_optimize chain}

\begin{frame}{From a sentence to an optimised flow}
  \begin{center}
  \begin{tikzpicture}[node distance=0.7cm and 0.9cm,
                      every node/.style={font=\footnotesize}]
    \node[notebox, minimum width=3.6cm] (spec) {natural-language spec\\\scriptsize ``Goertzel FP32 at 25\,MHz''};
    \node[agent2, right=of spec, minimum width=3.6cm] (idea)
      {\cmd{idea\_loop}\\\scriptsize sim-in-the-loop, critique feedback\\\scriptsize budget = turns};
    \node[agent3, right=of idea, minimum width=3.6cm,
          line width=0.8mm] (auto)
      {\cmd{autoresearch}\\\scriptsize propose / harden / sim / FoM\\\scriptsize budget = evals};
    \node[notebox, below=of idea, minimum width=3.6cm] (pi)
      {\cmd{phase\_idea/}\\\scriptsize \cmd{config.yaml}, \cmd{src/}, \cmd{tb/}, \cmd{runs/}};
    \node[notebox, below=of auto, minimum width=3.6cm] (po)
      {\cmd{phase\_optimize/}\\\scriptsize \cmd{results.tsv}, \cmd{plots/}};
    \draw[arrow] (spec) -- (idea);
    \draw[arrow] (idea) -- (auto);
    \draw[arrow] (idea) -- (pi);
    \draw[arrow] (auto) -- (po);
  \end{tikzpicture}
  \end{center}
  \footnotesize
  \begin{itemize}
    \item \cmd{idea\_loop}: one-shot RTL synthesis with sim and
          flow critique between turns; cocotb + iverilog drive the
          gate.
    \item \cmd{autoresearch}: greedy iteration over flow knobs ON
          TOP of phase 1's converged RTL; LibreLane drives the gate.
    \item \warn{Phase 2 runs unconditionally}: partial or blocked
          phase-1 outputs still feed the optimiser.
  \end{itemize}
  \note{%
    Two budgets, two definitions of progress. \cmd{idea\_loop}
    counts \emph{turns} -- each turn is the agent re-reading the
    last failure and trying a new RTL revision until sim and lint
    are green. \cmd{autoresearch} counts \emph{evals} -- each eval
    is a full LibreLane flow on a fixed RTL with new flow knobs.
    \cmd{--skip-phase-idea} reuses an already-converged
    \cmd{phase\_idea/} directory so we can re-optimise without
    burning idea-loop budget. The two phases share nothing
    architecturally; the contract is just the directory layout.}
\end{frame}
